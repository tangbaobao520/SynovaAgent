export function isRetryable(code: ErrorCodeType): boolean {
  switch (code) {
    case ErrorCode.TIMEOUT:
    case ErrorCode.NETWORK:
    case ErrorCode.RATE_LIMITED:
    case ErrorCode.PROVIDER_UNAVAILABLE:
    case ErrorCode.CONTEXT_OVERFLOW:    // 压缩后可重试
    case ErrorCode.PAYLOAD_TOO_LARGE:   // 压缩后可重试
      return true;
    case ErrorCode.AUTH_FAILED:          // 可重试（换凭据）
      return true;
    case ErrorCode.AUTH_PERMANENT:       // 不可重试，需人工介入
    case ErrorCode.BILLING_EXCEEDED:     // 不可重试
    case ErrorCode.MODEL_NOT_FOUND:
    case ErrorCode.INVALID_INPUT:
    case ErrorCode.INTERNAL:
      return false;
  }
}

// 修改 callWithRetry()——根据错误类型选择不同降级路径
export async function callWithRetry<T>(
  fn: () => Promise<T>,
  options: { maxRetries?: number; onError?: (err: DiagnosticAgentError) => void } = {},
): Promise<T> {
  for (let attempt = 0; attempt <= (options.maxRetries ?? 3); attempt++) {
    try {
      return await fn();
    } catch (err: any) {
      const error = err instanceof DiagnosticAgentError ? err : new DiagnosticAgentError(ErrorCode.INTERNAL, err.message, 0, true);

      // 根据错误类型选择策略
      if (error.code === ErrorCode.CONTEXT_OVERFLOW) {
        // 触发上下文压缩然后重试
        options.onError?.(error);
        continue;
      }
      if (error.code === ErrorCode.AUTH_PERMANENT) {
        throw error; // 不再重试
      }
      if (error.code === ErrorCode.RATE_LIMITED) {
        // 更长的退避
        await sleep(calculateBackoff(attempt, 5_000, 120_000));
        continue;
      }
      // 默认退避
      if (error.retryable && attempt < (options.maxRetries ?? 3)) {
        await sleep(calculateBackoff(attempt));
        continue;
      }
      throw error;
    }
  }
}

// 在 ToolDefinition 中加入操作分类
export interface ToolDefinition {
  name: string;
  description: string;
  parameters: ToolSchema;
  executionMode?: ToolExecutionMode;

  // 新增：操作类型（用于并行门控）
  operationType?: 'read' | 'write' | 'admin' | 'interactive';
  // 新增：副作用声明
  sideEffects?: 'none' | 'mutating' | 'destructive';
  // 新增：资源路径（用于路径冲突检测）
  resourcePath?: string;

  handler: (params: Record<string, unknown>) => Promise<Record<string, unknown>>;
}

// 新增：并行门控
export class ParallelGate {
  // 规则引擎
  canParallelize(tools: ToolDefinition[]): boolean {
    if (tools.length <= 1) return false;

    for (const t of tools) {
      // 交互式工具不能并行
      if (t.operationType === 'interactive') return false;
      // 破坏性操作不能并行
      if (t.sideEffects === 'destructive') return false;
    }

    // 所有只读工具可以并行
    if (tools.every(t => t.operationType === 'read')) return true;

    // 写操作检查路径冲突
    const writeTools = tools.filter(t => t.operationType === 'write');
    if (writeTools.length > 1) {
      const paths = writeTools.map(t => t.resourcePath).filter(Boolean);
      if (new Set(paths).size !== paths.length) return false; // 路径冲突
    }

    return true;
  }
}

// 在 ToolRegistry 中增加并行执行
async executeParallel(tools: Array<{name: string; params: Record<string, unknown>}>): Promise<Map<string, ToolCallResult>> {
  const defs = tools.map(t => this.tools.get(t.name)).filter(Boolean) as ToolDefinition[];
  if (!this.gate.canParallelize(defs)) {
    // 回退到顺序执行
    const results = new Map<string, ToolCallResult>();
    for (const t of tools) {
      results.set(t.name, await this.execute(t.name, t.params));
    }
    return results;
  }

  // 并行执行（最多 8 并发）
  const pool = new Map<string, Promise<ToolCallResult>>();
  for (const t of tools) {
    pool.set(t.name, this.execute(t.name, t.params));
  }
  const entries = await Promise.all(pool.values());
  const results = new Map<string, ToolCallResult>();
  tools.forEach((t, i) => results.set(t.name, entries[i]));
  return results;
}

// 新增：工具循环防护
export class ToolGuardrails {
  // 精确失败计数：相同工具+相同参数连续失败
  private exactFailCount = new Map<string, number>();
  // 相同工具失败计数：同一工具连续失败（不论参数）
  private sameToolFailCount = new Map<string, number>();
  // 结果哈希缓存：只读工具返回相同结果
  private resultHashes = new Map<string, string[]>();
  // 上次结果哈希（用于幂等检测）
  private lastResultHash = new Map<string, string>();

  readonly EXACT_FAIL_WARN = 2;    // 精确重复 2 次 → 警告
  readonly EXACT_FAIL_BLOCK = 5;   // 精确重复 5 次 → 阻止
  readonly SAME_TOOL_WARN = 3;     // 同工具 3 次 → 警告
  readonly SAME_TOOL_HALT = 8;     // 同工具 8 次 → 停止
  readonly NO_PROGRESS_BLOCK = 5;  // 幂等重复 5 次 → 阻止

  // 检查工具调用是否安全
  check(name: string, params: Record<string, unknown>, result: Record<string, unknown>): GuardrailDecision {
    const key = `${name}:${JSON.stringify(params)}`;
    const hash = this.sha256(JSON.stringify(result));

    // 1. 精确失败检测
    if (result.error) {
      this.exactFailCount.set(key, (this.exactFailCount.get(key) || 0) + 1);
      const count = this.exactFailCount.get(key)!;
      if (count >= this.EXACT_FAIL_BLOCK) return { action: 'block', reason: `工具 ${name} 相同参数连续失败 ${count} 次` };
      if (count >= this.EXACT_FAIL_WARN) return { action: 'warn', reason: `工具 ${name} 连续失败 ${count} 次` };
    }

    // 2. 幂等无进展检测（只读工具）
    const prevHash = this.lastResultHash.get(name);
    if (prevHash && prevHash === hash) {
      const history = this.resultHashes.get(name) || [];
      history.push(hash);
      this.resultHashes.set(name, history);
      if (history.length >= this.NO_PROGRESS_BLOCK) {
        return { action: 'block', reason: `工具 ${name} 连续 ${history.length} 次返回相同结果` };
      }
    }
    this.lastResultHash.set(name, hash);

    return { action: 'allow' };
  }

  // 回合开始时重置
  resetForTurn(): void {
    this.exactFailCount.clear();
    this.sameToolFailCount.clear();
    this.resultHashes.clear();
    this.lastResultHash.clear();
  }

  private sha256(input: string): string {
    const crypto = require('crypto');
    return crypto.createHash('sha256').update(input).digest('hex');
  }
}

export interface GuardrailDecision {
  action: 'allow' | 'warn' | 'block' | 'halt';
  reason?: string;
}

// 新增：凭据池（多凭据轮换）
export class CredentialPool {
  private credentials: Map<string, {
    connectorId: string;
    credentials: Record<string, string>;
    status: 'ok' | 'exhausted' | 'dead';
    lastError?: string;
    usedCount: number;
  }> = new Map();

  register(connectorId: string, creds: Record<string, string>): void {
    this.credentials.set(connectorId, {
      connectorId,
      credentials: creds,
      status: 'ok',
      usedCount: 0,
    });
  }

  // 轮询获取下一个可用凭据（round-robin）
  acquire(): { connectorId: string; credentials: Record<string, string> } | null {
    const available = [...this.credentials.values()]
      .filter(c => c.status === 'ok')
      .sort((a, b) => a.usedCount - b.usedCount);

    if (available.length === 0) return null;

    const selected = available[0];
    selected.usedCount++;
    return { connectorId: selected.connectorId, credentials: { ...selected.credentials } };
  }

  // 标记凭据状态
  markError(connectorId: string, error: string): void {
    const entry = this.credentials.get(connectorId);
    if (!entry) return;
    entry.lastError = error;
    entry.status = 'exhausted';
    // 24 小时后自动恢复
    setTimeout(() => { entry.status = 'ok'; }, 24 * 3600_000);
  }

  markDead(connectorId: string): void {
    const entry = this.credentials.get(connectorId);
    if (entry) entry.status = 'dead';
  }
}

async compact(threshold: number = 0.5): Promise<CompactionResult> {
  let summary: string | null = null;
  const originalLength = this.messages.length;

  // 层级 1：LLM 摘要（需要 LLM Provider）
  try {
    summary = await this.llmSummarize(this.messages);
  } catch (err) {
    log.warn({ err }, 'LLM 摘要失败，降级到确定性压缩');
  }

  // 层级 2：确定性摘要（保留标题、关键数据）
  if (!summary) {
    try {
      summary = this.deterministicSummarize(this.messages);
    } catch (err) {
      log.warn({ err }, '确定性摘要也失败，保留所有消息');
      return {
        compacted: false,
        originalLength,
        newLength: originalLength,
        summary: null,
      };
    }
