2. 专家自主权引擎的"幻觉"防御

2.1 当前实现的问题

ExpertAutonomyEngine 的 executeQuery 方法：

private async executeQuery(functionName: string): Promise<unknown> {
  switch (functionName) {
    case 'findDiagnosticPaths': return this.queryApi.findDiagnosticPaths('Risk', 'Person');
    case 'summarizeSubgraph':   return this.queryApi.summarizeSubgraph('root', 3);
    case 'findCrossDimensionalBrokers': return this.queryApi.findCrossDimensionalBrokers();
    case 'match_pattern':       return this.queryApi.matchPattern ? this.queryApi.matchPattern([]) : ...;
    default:                    return { error: \`未知查询: \${functionName}\` };
  }
}

三个严重问题：

  1. 参数硬编码：'findDiagnosticPaths' 总是查 'Risk'→'Person'，但 Expert 可能想查 'Team'→'Goal'
  2. 未知查询 = 错误返回给 LLM：LLM 收到错误后可能重新尝试或乱说，没有降级信号传递给上层
  3. 无 SOG Schema 验证：LLM 生成一个 'query_graph' → 'deleteAllNodes' 无法被阻止，因为没有 schema 感知的查询白名单

2.3 推荐的动态参数注入

// 让 Expert LLM 生成"参数化查询"而非"函数名"
interface ExpertQuery {
  action: 'find_path' | 'summarize' | 'find_brokers' | 'match_pattern';
  parameters: {
    fromType?: string;   // e.g., 'Risk'
    toType?: string;     // e.g., 'Person'
    rootId?: string;     // e.g., 'team-123'
    maxDepth?: number;   // e.g., 3
    signals?: string[];  // e.g., ['high_turnover', 'low_engagement']
  };
}

// 验证 + 执行
async executeQuery(raw: string): Promise<QueryResult> {
  const query: ExpertQuery = JSON.parse(raw);
  const validation = this.schemaValidator.validate(query);

  if (!validation.valid) {
    return {
      success: false,
      error: \`无效查询: \${validation.reason}\`,
      alternatives: validation.suggestions,  // 给 LLM 的替代方案
    };
  }

  // 安全执行（参数化，非字符串拼接）
  return this.queryApi.safeExecute(validation.normalized);
}

4.2 混合推理框架

// "通用规则" + "专家规则" 混合推理
interface Rule {
  id: string;
  type: 'general' | 'expert';
  condition: (evidence: Evidence[], graph: GraphStoreRO) => boolean;
  conclusion: Hypothesis;
  confidence: number;  // 0-1, 初始值：general=0.5, expert=0.7
}

class HybridReasoner {
  private generalRules: Rule[] = [
    // 组织理论基本公理
    { id: 'span_of_control', condition: (e) => e.ratio('manager:ic') > 1:15,
      conclusion: { type: 'org_design', severity: 'medium' }, confidence: 0.6 },
    // Conway 定律
    { id: 'conway_law', condition: (e) => e.structMismatch('team', 'system'),
      conclusion: { type: 'architecture', severity: 'high' }, confidence: 0.7 },
  ];

  private expertRules: Rule[] = []; // 由行业专家贡献，动态加载

  async reason(evidence: Evidence[]): Promise<Hypothesis[]> {
    const results: Hypothesis[] = [];
    for (const rule of [...this.generalRules, ...this.expertRules]) {
      if (rule.condition(evidence, this.graph)) {
        results.push({ ...rule.conclusion, confidence: rule.confidence });
      }
    }
    return results;
  }
}

5. 实体解析优化（中文 Jaccard 问题）

铁律 29 被违反：entity-resolver.ts 使用 Jaccard token overlap 计算中文姓名相似度。Jaccard 对中文的分词基于单字符（因为 tokenize 正则按单个汉字切分），导致"张三"和"李四"的 Jaccard 相似度为 0（完全不同），但"张三"和"张三丰"的 Jaccard 相似度为 0.5（共享"张三"字符）。人名解析完全不可靠。

5.1 推荐的替代方案

方法精度速度实现复杂度推荐
LLM-as-Judge高慢（1-3s/对）低🥇 边界案例（0.65-0.85 区间）
拼音编码 + 编辑距离中快（<1ms）低🥇 中文姓名主方案
Sentence-Transformers高中（10-50ms）中（需引入 ONNX 模型）🥈 通用文本方案
字符 n-gram + 加权 Jaccard中快低🥉 快速改进

5.2 拼音编码方案

import { pinyin } from 'pinyin-pro'; // 10KB 的拼音库

function phoneticSimilarity(nameA: string, nameB: string): number {
  // 将中文转为拼音（忽略声调）
  const pyA = pinyin(nameA, { toneType: 'none' });
  const pyB = pinyin(nameB, { toneType: 'none' });

  // 计算编辑距离（Levenshtein）
  const distance = levenshteinDistance(pyA, pyB);
  const maxLen = Math.max(pyA.length, pyB.length);

  // 归一化为相似度
  return 1 - (distance / maxLen);
}

// "王芳" → "wang fang"
// "王放" → "wang fang"
// 拼音相似度 ≈ 1.0（同音字），文本 Jaccard = 0.5（不同字）

6.3 闭环补全方案

// 完整的验证闭环（需要持久化）
class AutomatedValidationLoop {
  async run(): Promise<void> {
    // Step 1: 检查已采纳的建议，追踪实际改善
    const adopted = await this.fetchAdoptedRecommendations();
    for (const rec of adopted) {
      const outcome = await this.measureOutcome(rec);
      await this.recordOutcome(rec.id, outcome);

      // Step 2: 根据结果调整模式权重
      if (outcome.improved) {
        await this.patternEngine.boostWeight(rec.patternId, 0.1);
        await this.templateValidator.recordValidation(rec.patternId, {
          confirmed: true,
          outcomeScore: outcome.score,
        });
      } else {
        await this.patternEngine.reduceWeight(rec.patternId, 0.1);
        if (outcome.score < 0.3) {
          await this.templateValidator.markOutdated(rec.patternId);
        }
      }
    }
  }

  // Step 3: 周期性检查模式是否仍然相关
  async stalenessCheck(): Promise<void> {
    const activeTemplates = await this.templateValidator.getActiveTemplates();
    for (const tpl of activeTemplates) {
      if (this.stalenessDetector.isStale(tpl.id)) {
        await this.templateValidator.requestReview(tpl.id);
      }
    }
  }
}
