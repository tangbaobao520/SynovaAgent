LLM Provider 扩展方案

国内主流模型适配 + 海外模型扩展 · 2026-06-05

架构决策

所有国产 LLM（通义千问/智谱GLM/Kimi/零一万物/MiniMax/阶跃星辰/文心一言）都使用 OpenAI 兼容 API。所以我们不需要为每家写独立的 Provider 实现——已有的 openai.ts 适配器可以直接复用。需要补充的是：

  • Provider 类型枚举 + 配置默认值（baseUrl / model）
  • Provider 自动检测逻辑（从环境变量推断）
  • 健康检查端点适配
  • Provider 标签/说明

文心一言特殊说明

文心一言的 API 不是标准 OpenAI 兼容格式（使用 access_token 而非 Bearer token）。需要独立的 ernie.ts 适配器。
其他所有国产模型都是 OpenAI 兼容格式，复用 openai.ts 即可。

三、index.ts 修改样例

export type ProviderType = 'deepseek' | 'qwen' | 'glm' | 'kimi' | 'yi' | 'minimax' | 'step' | 'ernie' | 'openai' | 'gateway';

const PROVIDER_LABELS: Record<ProviderType, string> = {
  deepseek: 'DeepSeek V4 (默认, 国内低延迟, 1M 上下文)',
  qwen: '通义千问 (阿里云)',
  glm: '智谱 GLM',
  kimi: 'Kimi (Moonshot)',
  yi: '零一万物 Yi',
  minimax: 'MiniMax',
  step: '阶跃星辰 Step',
  ernie: '文心一言 (百度)',
  openai: 'OpenAI / 海外通用',
  gateway: '自定义 Gateway',
};

export function createProvider(type: ProviderType, config: ProviderConfig): LLMProvider {
  switch (type) {
    case 'deepseek': return createDeepSeekProvider(config);
    case 'qwen':     return createOpenAIProvider({ ...config, baseUrl: config.baseUrl || 'https://dashscope.aliyuncs.com/compatible-mode/v1', model: config.model || 'qwen-plus' });
    case 'glm':      return createOpenAIProvider({ ...config, baseUrl: config.baseUrl || 'https://open.bigmodel.cn/api/paas/v4', model: config.model || 'glm-4-plus' });
    case 'kimi':     return createOpenAIProvider({ ...config, baseUrl: config.baseUrl || 'https://api.moonshot.cn/v1', model: config.model || 'moonshot-v1-8k' });
    case 'yi':       return createOpenAIProvider({ ...config, baseUrl: config.baseUrl || 'https://api.lingyiwanwu.com/v1', model: config.model || 'yi-lightning' });
    case 'minimax':  return createOpenAIProvider({ ...config, baseUrl: config.baseUrl || 'https://api.minimax.chat/v1', model: config.model || 'minimax-text-01' });
    case 'step':     return createOpenAIProvider({ ...config, baseUrl: config.baseUrl || 'https://api.stepfun.com/v1', model: config.model || 'step-1-8k' });
    case 'ernie':    return createErnieProvider(config);
    case 'openai':   return createOpenAIProvider(config);
    case 'gateway':  return createGatewayProvider(config);
    default:         throw new Error(`不支持的 Provider: ${type}`);
  }
}

四、detect.ts 修改样例

export function detectProvider(): ProviderType {
  if (process.env.OPENCLAW_GATEWAY_HOST) return 'gateway';
  if (process.env.ERNIE_API_KEY) return 'ernie';
  if (process.env.QWEN_API_KEY || process.env.DASHSCOPE_API_KEY) return 'qwen';
  if (process.env.GLM_API_KEY || process.env.ZHIPU_API_KEY) return 'glm';
  if (process.env.MOONSHOT_API_KEY || process.env.KIMI_API_KEY) return 'kimi';
  if (process.env.YI_API_KEY || process.env.LINGYI_API_KEY) return 'yi';
  if (process.env.MINIMAX_API_KEY) return 'minimax';
  if (process.env.STEP_API_KEY) return 'step';
  if (process.env.OPENAI_API_KEY) return 'openai';
  return 'deepseek'; // 默认
}

export interface SynovaConfig {

  llmApiKey: string;

  // 各 Provider API Key（自动检测）
  deepseekKey: string;
  qwenKey: string;
  glmKey: string;
  kimiKey: string;
  yiKey: string;
  minimaxKey: string;
  stepKey: string;
  ernieKey: string;
  ernieSecretKey: string;
  openaiKey: string;
}

export function loadConfig(): SynovaConfig {
  const provider = detectProvider();
  const providerKey = resolveApiKey(provider);

  return {
    llmProvider: provider,
    llmApiKey: providerKey,
    llmBaseUrl: resolveBaseUrl(provider, process.env),
    llmModel: resolveModel(provider, process.env),
    // ...
  };
}
