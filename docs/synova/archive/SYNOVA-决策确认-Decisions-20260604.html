# synova_worker/connectors/base.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any, Dict, List, Optional

@dataclass
class ConnectorConfig:
    name: str
    platform: str
    credentials: Dict[str, str]
    org_id: str

class ConnectorBase(ABC):
    @abstractmethod
    def health_check(self) -> Dict[str, Any]:
        """验证连接是否可用"""
        pass

    @abstractmethod
    def fetch_members(self) -> List[Dict[str, Any]]:
        """获取组织成员列表"""
        pass

    @abstractmethod
    def fetch_messages(self, since: str, until: Optional[str] = None) -> List[Dict[str, Any]]:
        """获取协作消息"""
        pass

    @abstractmethod
    def map_to_sog(self, data: Dict[str, Any]) -> Dict[str, Any]:
        """将平台数据映射为 SOG JSON（nodes + edges）"""
        pass

实体解析最终方案

// Phase 1（本次交付做）：纯 TypeScript 拼音编码
import { pinyin } from 'pinyin-pro';  // 10KB

function phoneticSimilarity(nameA: string, nameB: string): number {
  // "张翠山" → "zhang cui shan"
  // "张翠珊" → "zhang cui shan"
  // 拼音相同 → 高相似度（同音字匹配）
  const pyA = pinyin(nameA, { toneType: 'none' }).join(' ');
  const pyB = pinyin(nameB, { toneType: 'none' }).join(' ');
  const dist = levenshteinDistance(pyA, pyB);
  return 1 - dist / Math.max(pyA.length, pyB.length, 1);
}
