  constructor() {

    this.registerNodeType('Team', { category: 'actor', color: '#45b7d1' });
    // ...
  }

  // 运行时注册（扩展调用的入口）
  registerNodeType(name: string, schema: NodeTypeSchema): void {
    this.nodeTypes.set(name, schema);
  }

  registerEdgeType(name: string, schema: EdgeTypeSchema): void {
    this.edgeTypes.set(name, schema);
  }

  // 验证（运行时注册的类型可以通过验证）
  validateNodeType(type: string): boolean {
    return this.nodeTypes.has(type) || this.isValidDynamicType(type);
  }
}

3. 联邦进化攻击向量分析

联邦进化是产品战略级能力，但目前零实现。以下分析假设我们实现了基本的联邦学习框架，分析其安全风险。

3.1 威胁模型

攻击者类型能力目标严重性
恶意客户端控制一个联邦节点污染全局模型（投毒攻击）高
窃听者拦截联邦通信窃取其他组织的诊断模式高（隐私泄露）
恶意服务器控制联邦中心推断客户端数据高（隐私泄露）
拜占庭客户端控制多个联邦节点控制全局聚合结果严重

3.2 防御机制对比

防御机制防御的攻击隐私保证精度损失实现复杂度
差分隐私（DP）模型反推、成员推理ε-DP 保证中（ε=8 时 ~5%）低（加噪即可）
安全聚合（SecAgg）窃听者、恶意服务器计算安全无高（密码学协议）
拜占庭容错聚合恶意/拜占庭客户端无低（~1-3%）中（Krum/Trimmed Mean）
零知识证明恶意客户端计算安全无极高

3.3 推荐的防御架构

// 分层防御：差分隐私 + 拜占庭容错 + 安全聚合

// Layer 1: 本地差分隐私
function addLocalDP(gradient: number[], epsilon: number, sensitivity: number): number[] {
  const scale = sensitivity / epsilon;
  return gradient.map(g => g + laplaceNoise(0, scale));
  // ε = 8: 对大多数企业可接受，噪声在可容忍范围内
}

// Layer 2: 拜占庭容错聚合（Krum 算法）
function byzantineRobustAggregate(updates: number[][], f: number): number[] {
  // f = 可容忍的拜占庭节点数
  // Krum: 选择与其他更新最"一致"的更新
  // 选择与其他更新欧式距离之和最小的更新
  const n = updates.length;
  if (n < 2 * f + 3) throw new Error('节点数不足以防御拜占庭攻击');

  const scores = updates.map((u1, i) => {
    const distances = updates
      .filter((_, j) => j !== i)
      .map(u2 => euclideanDistance(u1, u2))
      .sort((a, b) => a - b);
    // 取最近的 n-f-2 个距离之和
    return distances.slice(0, n - f - 2).reduce((a, b) => a + b, 0);
  });

  // 选择得分最小的更新（最"中心"的）
  const krumIndex = scores.indexOf(Math.min(...scores));
  return updates[krumIndex];
}

// Layer 3: 安全聚合（Shamir 秘密共享，简化版）
// 每个客户端将自己的更新分割为 n 个秘密共享
// 服务器收到至少 t 个共享后才能重建聚合

4. 本体自生长（Ontology Learning）

4.1 当前 L0 自适应

当前 evolution/l0-adaptation.ts 只做简单的"诊断发现 → 创建同名节点"：

  • 使用 >=0.7 置信度的诊断发现生成 OntologyPatch
  • 补丁只做 create 动作（无 update/delete）
  • 补丁不会发现「新的节点类型」——只会创建已有类型的节点

4.2 前沿 Ontology Learning 方法

方法描述适合我们的场景？
术语抽取（Term Extraction）从文本中抽取候选术语，聚类形成概念✅ 可以从诊断报告中提取新概念
关系发现（Relation Discovery）基于共现/模式发现概念间关系✅ 可从高频共同诊断发现新关系
LLM 驱动的本体提案用 LLM 阅读诊断报告，提案新类型✅ 最适合冷启动
概念漂移检测（Concept Drift）监控新出现的实体模式✅ 适合已运行的系统

4.3 推荐的 Ontology Self-Growth 流程

// Phase D (后诊断阶段) — 本体自生长

class OntologySelfGrowth {
  async proposeNewTypes(orgId: string): Promise<TypeProposal[]> {
    // Step 1: 收集诊断中发现的关键词/实体
    const findings = await this.getRecentFindings(orgId);
    const keywords = await this.extractKeywords(findings);

    // Step 2: LLM 分析候选新类型
    const analysis = await this.llm.consult(
      `以下是组织诊断中反复出现的概念列表：
       ${keywords.join(', ')}

       其中哪些概念可以作为新的 SOG 节点类型？
       对新类型给出：
       - 名称（英文驼峰）
       - 描述
       - 预期的邻居关系
       - 示例

       当前已有类型：${SOG_NODE_TYPES.join(', ')}`,
    );

    // Step 3: 格式化提案
    const proposals = this.parseProposals(analysis);

    // Step 4: 人工审核队列
    await this.reviewQueue.enqueue(proposals.map(p => ({
      ...p,
      status: 'pending_review',
      confidence: this.calculateConfidence(p),
    })));

    return proposals;
  }

  // Step 5: 管理员批准后 → 注册新类型
  async approveNewType(proposal: TypeProposal): Promise<void> {
    this.schemaRegistry.registerNodeType(proposal.name, proposal.schema);
    await this.auditLog.record({
      action: 'schema_evolution',
      detail: \`添加新节点类型: \${proposal.name}\`,
    });
  }
}
