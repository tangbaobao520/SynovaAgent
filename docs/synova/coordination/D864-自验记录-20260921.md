// D864 自验 P2-b 独立反例（p1b-verifier 自拟）
import { describe, it, expect, vi } from 'vitest';
vi.mock('../../src/providers/http-exit', () => ({
  getProxyStatus: () => { throw new Error('proxy boom http://verifier-user:top-secret-d864-pass@10.9.9.9:3128'); },
}));
const express = (await import('express')).default;
const healthz = (await import('../../src/routes/healthz')).default;
describe('D864 自验反例：checkOutboundProxy 异常不泄值（整包断言）', () => {
  it('mock 抛含凭据异常 → 整包 JSON 不含凭据片段', async () => {
    const app = express(); app.use(healthz);
    const srv = app.listen(0);
    const { port } = srv.address() as { port: number };
    try {
      const res = await fetch(`http://127.0.0.1:${port}/api/healthz`);
      const whole = await res.text();
      for (const frag of ['top-secret-d864-pass', 'verifier-user', 'user:', '10.9.9.9', '3128']) {
        expect(whole.includes(frag), `整包不得含 ${frag}; body=${whole}`).toBe(false);
      }
      const body = JSON.parse(whole) as { checks: Record<string, { status: string }> };
      expect(body.checks.outbound_proxy.status).toBe('degraded');
    } finally { await new Promise((r) => srv.close(r)); }
  });
});
