一、统一数据格式（先做这个）

所有适配器上报的数据格式统一。在 src/agent-observer/types.ts 定义：

// 一个 Agent 活动事件
export interface AgentActivity {
  agentId: string;          // Agent 唯一标识
  platform: string;         // 'openclaw' | 'claude-code' | 'hermes' | 'langchain' | 'mcp' | 'unknown'
  agentName?: string;       // 可读名称
  teamId?: string;          // 所属团队（如果有）

  // 活动内容
  activityType: 'tool_call' | 'llm_call' | 'subagent_spawn' | 'task_complete' | 'error' | 'heartbeat';
  toolName?: string;        // 如果是工具调用
  toolArgs?: string;         // 工具参数（可选，默认不发送）
  success?: boolean;        // 是否成功
  durationMs?: number;       // 耗时
  error?: string;            // 错误信息

  // 资源
  tokenIn?: number;          // LLM 输入 token
  tokenOut?: number;         // LLM 输出 token
  costUsd?: number;          // 成本（美元）

  // 时间
  timestamp: string;         // ISO 8601
  sessionId?: string;        // 会话 ID（如果有）
}

// 上报 API
// POST /api/agent-observer/report
// Body: AgentActivity 或 AgentActivity[]
// 响应: { ok: true, nodeId?: string }

二、接收端（Synova 侧）

新建 src/agent-observer/collector.ts，接收上报数据并写入 SOG：

import { getGraphStore } from '../l4/graph-bridge';
import { SOGNodeType, SOGEdgeType } from '@synova/sog-core';
import { createLogger } from '../logger';
import type { AgentActivity } from './types';

const log = createLogger('agent-observer/collector');

export async function collectActivity(activity: AgentActivity): Promise<{ nodeId: string }> {
  const store = getGraphStore();
  const graph = activity.teamId || 'default';
  const nodeId = `agent-${activity.platform}-${activity.agentId}`;

  // 创建或更新 AGENT 节点
  store.createNode(SOGNodeType.AGENT, {
    name: activity.agentName || activity.agentId,
    platform: activity.platform,
    lastSeen: activity.timestamp,
    status: 'active',
  }, graph);

  // 记录工具调用等信息（可选）
  if (activity.activityType === 'tool_call' && activity.success !== undefined) {
    // 可以记录到 EVENT 节点或更新节点属性
  }

  return { nodeId };
}

新建 src/routes/agent-observer.ts：

import { Router } from 'express';
import { collectActivity } from '../agent-observer/collector';
import type { AgentActivity } from '../agent-observer/types';

const router = Router();

router.post('/api/agent-observer/report', async (req, res) => {
  try {
    const activities = Array.isArray(req.body) ? req.body : [req.body];
    const results = await Promise.allSettled(
      activities.map(a => collectActivity(a as AgentActivity))
    );
    const nodeIds = results
      .filter((r): r is PromiseFulfilledResult<{ nodeId: string }> => r.status === 'fulfilled')
      .map(r => r.value.nodeId);
    res.json({ ok: true, nodeIds, count: nodeIds.length });
  } catch (err: any) {
    res.status(500).json({ ok: false, error: err.message });
  }
});

export default router;

在 src/server.ts 中挂载此路由。

接线验证：curl -X POST http://localhost:3000/api/agent-observer/report -H 'Content-Type: application/json' -d '{"agentId":"test","platform":"openclaw","activityType":"heartbeat","timestamp":"2026-06-05T00:00:00Z"}' → 返回 200

三、MCP Server 适配器（覆盖 10 个框架，P0）

这些框架的共性：都是配置一个 MCP Server 在 JSON 文件里，不需要写代码。

新建 packages/agent-observer-mcp/src/index.ts（独立的 npm 包，@synova/agent-observer-mcp）：

import { Server } from '@modelcontextprotocol/sdk/server/index.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import fetch from 'node-fetch';

const SYNOVA_URL = process.env.SYNOVA_URL || 'http://localhost:3000';
const AGENT_NAME = process.env.AGENT_NAME || 'unknown-agent';
const PLATFORM = 'mcp';

const server = new Server(
  { name: 'synova-observer', version: '1.0.0' },
  { capabilities: { tools: {} } }
);

// 提供一个工具供 Agent 调用（Agent 主动汇报）
server.setRequestHandler('tools/call', async (request) => {
  const { name, arguments: args } = request.params;

  // 上报到 Synova
  await fetch(`${SYNOVA_URL}/api/agent-observer/report`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      agentId: AGENT_NAME,
      platform: PLATFORM,
      agentName: AGENT_NAME,
      activityType: 'tool_call',
      toolName: name,
      toolArgs: JSON.stringify(args).slice(0, 1000),
      timestamp: new Date().toISOString(),
    }),
  }).catch(() => {}); // 静默失败，不影响 Agent

  return { content: [{ type: 'text', text: 'ok' }] };
});

const transport = new StdioServerTransport();
await server.connect(transport);

"""HTTP 上报到 Synova（所有适配器共用）"""
import requests
import json
import os

SYNOVA_URL = os.environ.get('SYNOVA_URL', 'http://localhost:3000')

def report(activity: dict) -> None:
    """上报 Agent 活动到 Synova。失败时不抛异常。"""
    try:
        requests.post(
            f"{SYNOVA_URL}/api/agent-observer/report",
            json=activity,
            timeout=3,
        )
    except Exception:
        pass  # 静默失败，不影响 Agent

新建 observer-adapters/openclaw-skill/report.py：

#!/usr/bin/env python3
"""OpenClaw agent_end 钩子：采集 Agent 活动上报到 Synova"""
import sys, json
from synova_worker.observer.report import report

# 从 stdin 读取钩子事件（OpenClaw 的钩子系统通过 stdin/stdout 通信）
input_data = json.loads(sys.stdin.read())
hook_event = input_data.get('hookEvent', {})
agent_context = input_data.get('context', {})

# 提取活动数据
activity = {
    'agentId': agent_context.get('agentId', 'unknown'),
    'platform': 'openclaw',
    'agentName': agent_context.get('agentName', agent_context.get('agentId', '')),
    'teamId': agent_context.get('teamId', ''),
    'activityType': 'tool_call',
    'success': hook_event.get('success', True),
    'durationMs': hook_event.get('durationMs', 0),
    'timestamp': hook_event.get('timestamp', ''),
}
report(activity)

# 输出空 JSON 表示成功（OpenClaw 钩子协议要求）
print(json.dumps({}))

#!/usr/bin/env python3
"""Claude Code PreToolUse 钩子：采集工具调用上报到 Synova"""
import sys, json
from synova_worker.observer.report import report

input_data = json.loads(sys.stdin.read())
tool_name = input_data.get('tool_name', '')
tool_input = input_data.get('tool_input', {})

activity = {
    'agentId': 'claude-code',
    'platform': 'claude-code',
    'activityType': 'tool_call',
    'toolName': tool_name,
    'toolArgs': json.dumps(tool_input).slice(0, 1000),
    'timestamp': '',

report(activity)

# 允许工具继续执行
print(json.dumps({}))

"""Hermes Plugin：采集 Agent 活动上报到 Synova"""
from synova_worker.observer.report import report

def register(api):
    """在 Hermes Plugin 系统中注册"""
    @api.on('tool_call_completed')
    def on_tool_call(tool_name, args, result, ctx):
        report({
            'agentId': ctx.agent_id,
            'platform': 'hermes',
            'agentName': ctx.agent_name,
            'activityType': 'tool_call',
            'toolName': tool_name,
            'success': result.success,
            'durationMs': result.duration_ms,
            'timestamp': ctx.timestamp.isoformat(),
        })

from synova_worker.observer.report import report

class SynovaCallback:
    """LangChain CallbackHandler / CrewAI callback / 通用"""
    def on_tool_start(self, tool_name, **kwargs):
        report({
            'agentId': kwargs.get('agent_id', 'python-agent'),
            'platform': 'langchain',
            'activityType': 'tool_call_start',
            'toolName': tool_name,
            'timestamp': '',
        })

    def on_tool_end(self, tool_name, output, **kwargs):
        report({
            'agentId': kwargs.get('agent_id', 'python-agent'),
            'platform': 'langchain',
            'activityType': 'tool_call_end',
            'toolName': tool_name,
            'success': True,
            'timestamp': '',
        })
